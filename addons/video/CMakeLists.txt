list(APPEND CMAKE_MODULE_PATH "${PROJECT_SOURCE_DIR}/cmake")
find_package(MPV REQUIRED)
add_library(cui_video STATIC cui_mpv.c)
add_library(cui::video ALIAS cui_video)
set_target_properties(cui_video PROPERTIES EXPORT_NAME video POSITION_INDEPENDENT_CODE ON)
target_link_libraries(cui_video PUBLIC cui::cui PRIVATE MPV::MPV)
target_compile_definitions(cui_video PRIVATE $<$<BOOL:${CUI_TEXT_SHAPING}>:CUI_TEXT_SHAPING>)
if(MSVC)
    target_compile_options(cui_video PRIVATE /W4 /wd4100 /wd4244 /wd4267 /utf-8 /Gy /Gw)
else()
    target_compile_options(cui_video PRIVATE -Wall -Wextra -ffunction-sections -fdata-sections)
endif()
include(GNUInstallDirs)
install(TARGETS cui_video EXPORT cuiVideoTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(EXPORT cuiVideoTargets NAMESPACE cui:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/cui)
configure_file(${PROJECT_SOURCE_DIR}/cmake/cui-video.pc.in ${PROJECT_BINARY_DIR}/cui-video.pc @ONLY)
install(FILES ${PROJECT_BINARY_DIR}/cui-video.pc DESTINATION ${CMAKE_INSTALL_LIBDIR}/pkgconfig)
