# Native browser integration; the core never links platform browser libraries.
add_library(cui_webview STATIC cui_webview.c)
add_library(cui::webview ALIAS cui_webview)
set_target_properties(cui_webview PROPERTIES EXPORT_NAME webview POSITION_INDEPENDENT_CODE ON)
target_link_libraries(cui_webview PUBLIC cui::cui)
target_include_directories(cui_webview PRIVATE ${PROJECT_SOURCE_DIR}/src)
target_compile_definitions(cui_webview PRIVATE CUI_WEBVIEW $<$<BOOL:${CUI_TEXT_SHAPING}>:CUI_TEXT_SHAPING>)
if(MSVC)
    target_compile_options(cui_webview PRIVATE /W4 /wd4100 /wd4201 /wd4204 /wd4221 /wd4244 /wd4267 /wd4305 /wd4996 /utf-8 /Gy /Gw)
    target_compile_definitions(cui_webview PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
    target_compile_options(cui_webview PRIVATE -Wall -Wextra -Wno-unused-parameter -Wno-missing-field-initializers -ffunction-sections -fdata-sections)
endif()

if(CUI_BACKEND STREQUAL "cocoa")
    target_sources(cui_webview PRIVATE cui_webview_cocoa.m)
    set_source_files_properties(cui_webview_cocoa.m PROPERTIES COMPILE_FLAGS "-fobjc-arc")
    target_link_libraries(cui_webview PUBLIC "-framework WebKit")
elseif(CUI_BACKEND STREQUAL "win32")
    set(CUI_WEBVIEW2_SDK "" CACHE PATH "Extracted Microsoft.Web.WebView2 NuGet package")
    find_path(CUI_WEBVIEW2_INCLUDE_DIR WebView2.h HINTS "${CUI_WEBVIEW2_SDK}/build/native/include")
    if(NOT CUI_WEBVIEW2_INCLUDE_DIR)
        message(FATAL_ERROR "CUI_WEBVIEW needs WebView2.h; set CUI_WEBVIEW2_SDK to the extracted NuGet package")
    endif()
    target_include_directories(cui_webview SYSTEM PRIVATE ${CUI_WEBVIEW2_INCLUDE_DIR})
    if(MINGW)
        target_include_directories(cui_webview PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/webview2_compat)
    endif()
    if(CMAKE_SYSTEM_PROCESSOR MATCHES "ARM64|arm64|aarch64" OR CMAKE_GENERATOR_PLATFORM STREQUAL "ARM64")
        set(CUI_WEBVIEW2_ARCH arm64)
    elseif(CMAKE_SIZEOF_VOID_P EQUAL 8)
        set(CUI_WEBVIEW2_ARCH x64)
    else()
        set(CUI_WEBVIEW2_ARCH x86)
    endif()
    find_library(CUI_WEBVIEW2_LOADER NAMES WebView2Loader.dll
        HINTS "${CUI_WEBVIEW2_SDK}/build/native/${CUI_WEBVIEW2_ARCH}")
    if(NOT CUI_WEBVIEW2_LOADER)
        message(FATAL_ERROR "CUI_WEBVIEW needs the ${CUI_WEBVIEW2_ARCH} WebView2Loader.dll.lib from the SDK")
    endif()
    target_link_libraries(cui_webview PUBLIC ${CUI_WEBVIEW2_LOADER})
    target_sources(cui_webview PRIVATE cui_webview_win32.c)
elseif(CUI_BACKEND STREQUAL "x11")
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(CUI_WEBKIT REQUIRED IMPORTED_TARGET webkit2gtk-4.1)
    target_link_libraries(cui_webview PUBLIC PkgConfig::CUI_WEBKIT)
    target_sources(cui_webview PRIVATE cui_webview_gtk.c)
else()
    message(FATAL_ERROR "CUI_WEBVIEW requires the cocoa, win32 or x11 backend")
endif()

include(GNUInstallDirs)
install(TARGETS cui_webview EXPORT cuiWebviewTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(EXPORT cuiWebviewTargets NAMESPACE cui:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/cui)
set(CUI_WEBVIEW_PC_LIBS "-lcui_webview")
set(CUI_WEBVIEW_PC_REQUIRES "cui")
if(CUI_BACKEND STREQUAL "cocoa")
    string(APPEND CUI_WEBVIEW_PC_LIBS " -framework WebKit")
elseif(CUI_BACKEND STREQUAL "x11")
    string(APPEND CUI_WEBVIEW_PC_REQUIRES " webkit2gtk-4.1")
elseif(CUI_BACKEND STREQUAL "win32")
    string(APPEND CUI_WEBVIEW_PC_LIBS " \"${CUI_WEBVIEW2_LOADER}\"")
endif()
configure_file(${PROJECT_SOURCE_DIR}/cmake/cui-webview.pc.in ${PROJECT_BINARY_DIR}/cui-webview.pc @ONLY)
install(FILES ${PROJECT_BINARY_DIR}/cui-webview.pc DESTINATION ${CMAKE_INSTALL_LIBDIR}/pkgconfig)
