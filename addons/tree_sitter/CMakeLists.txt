# The core never links Tree-sitter, even when this add-on is available.
list(APPEND CMAKE_MODULE_PATH "${PROJECT_SOURCE_DIR}/cmake")
find_package(TreeSitter REQUIRED)
add_library(cui_tree_sitter STATIC cui_tree_sitter.c)
add_library(cui::tree_sitter ALIAS cui_tree_sitter)
set_target_properties(cui_tree_sitter PROPERTIES EXPORT_NAME tree_sitter POSITION_INDEPENDENT_CODE ON)
target_link_libraries(cui_tree_sitter PUBLIC cui::cui PRIVATE TreeSitter::TreeSitter)
target_compile_definitions(cui_tree_sitter PUBLIC CUI_TREE_SITTER)
if(MSVC)
    target_compile_options(cui_tree_sitter PRIVATE /W4 /wd4100 /wd4244 /wd4267 /utf-8 /Gy /Gw)
else()
    target_compile_options(cui_tree_sitter PRIVATE -Wall -Wextra -ffunction-sections -fdata-sections)
endif()
include(GNUInstallDirs)
install(TARGETS cui_tree_sitter EXPORT cuiTreeSitterTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(EXPORT cuiTreeSitterTargets NAMESPACE cui:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/cui)
configure_file(${PROJECT_SOURCE_DIR}/cmake/cui-tree-sitter.pc.in ${PROJECT_BINARY_DIR}/cui-tree-sitter.pc @ONLY)
install(FILES ${PROJECT_BINARY_DIR}/cui-tree-sitter.pc DESTINATION ${CMAKE_INSTALL_LIBDIR}/pkgconfig)
