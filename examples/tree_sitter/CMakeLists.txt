# Explicitly enabled demo only: no grammars enter cui::cui or cui::tree_sitter.
cmake_minimum_required(VERSION 3.18)
include(FetchContent)
if(POLICY CMP0135)
    cmake_policy(SET CMP0135 NEW)
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(cui_demo_generated "${CMAKE_CURRENT_BINARY_DIR}/generated")
set(cui_demo_manifest "${CMAKE_CURRENT_SOURCE_DIR}/languages.json")
set(cui_demo_generator "${CMAKE_CURRENT_SOURCE_DIR}/generate.py")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${cui_demo_manifest}" "${cui_demo_generator}")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${cui_demo_generator}" cmake
    "${cui_demo_manifest}" "${cui_demo_generated}" RESULT_VARIABLE cui_demo_result)
if(NOT cui_demo_result EQUAL 0)
    message(FATAL_ERROR "Could not generate the demo grammar manifest")
endif()
file(WRITE "${cui_demo_generated}/roots.txt" "")
function(cui_demo_grammar name cache url hash directory)
    FetchContent_Declare(cui_ts_${cache} URL "${url}" URL_HASH SHA256=${hash}
        SOURCE_SUBDIR cui-demo-no-cmake)
    FetchContent_MakeAvailable(cui_ts_${cache})
    set(root "${cui_ts_${cache}_SOURCE_DIR}/${directory}")
    file(APPEND "${cui_demo_generated}/roots.txt" "${name}\t${root}\n")
    add_library(cui_demo_ts_${name} STATIC "${root}/src/parser.c")
    set_target_properties(cui_demo_ts_${name} PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
    if(EXISTS "${root}/src/scanner.c")
        target_sources(cui_demo_ts_${name} PRIVATE "${root}/src/scanner.c")
    endif()
    target_include_directories(cui_demo_ts_${name} PRIVATE "${root}/src")
    target_link_libraries(tree_sitter PRIVATE cui_demo_ts_${name})
    set(cui_demo_root_${name} "${root}" PARENT_SCOPE)
endfunction()
function(cui_demo_alias name grammar)
    file(APPEND "${cui_demo_generated}/roots.txt" "${name}\t${cui_demo_root_${grammar}}\n")
endfunction()
add_executable(tree_sitter ../tree_sitter.c)
target_link_libraries(tree_sitter PRIVATE cui::tree_sitter TreeSitter::TreeSitter)
include("${cui_demo_generated}/grammars.cmake")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${cui_demo_generator}" registry
    "${cui_demo_manifest}" "${cui_demo_generated}" "${cui_demo_generated}/roots.txt"
    RESULT_VARIABLE cui_demo_result)
if(NOT cui_demo_result EQUAL 0)
    message(FATAL_ERROR "Could not generate the demo language registry")
endif()
configure_file(THIRD_PARTY_NOTICES.txt "${cui_demo_generated}/THIRD_PARTY_NOTICES.txt" COPYONLY)
target_sources(tree_sitter PRIVATE "${cui_demo_generated}/languages.c")
target_include_directories(tree_sitter PRIVATE "${cui_demo_generated}")
if(MSVC)
    target_compile_options(tree_sitter PRIVATE /utf-8)
endif()
cui_app_target(tree_sitter)
set_target_properties(tree_sitter PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/examples")
if(APPLE)
    set_target_properties(tree_sitter PROPERTIES MACOSX_BUNDLE OFF)
endif()
